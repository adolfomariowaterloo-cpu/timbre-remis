// Enviar mensaje de WhatsApp a uno o varios contactos (con ubicación opcional) function enviarWhatsAppMulti(contactos, mensajeBase, ubicacionUrl) { const mensajeFinal = ubicacionUrl ? `${mensajeBase}\n\n📍 Ubicación:\n${ubicacionUrl}` : `${mensajeBase}\n\n⚠️ No se pudo obtener ubicación. Por favor llamar.`; contactos.forEach(contacto => { if (contacto && Utils.validarNumero(contacto)) { const num = Utils.limpiarNumero(contacto); window.open(`https://wa.me/549${num}?text=${encodeURIComponent(mensajeFinal)}`, "_blank"); } }); }
